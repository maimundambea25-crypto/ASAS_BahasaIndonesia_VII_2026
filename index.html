☁️ Panduan Pemasangan Google Apps Script (Code.gs)

  1. Buka Google Spreadsheet baru, lalu klik menu Ekstensi > Apps Script.
  2. Hapus kode bawaan dan tempel kode di bawah ini, lalu klik Terapkan (Deploy) > Deployment Baru > Aplikasi Web.
  3. Pilih Yang memiliki akses: Siapa saja (Anyone), lalu salin URL Web App ke Panel Guru.
const SHEET_NAME = 'Rekap_LKS_CBT';
const TEACHER_PINS = ['011226', '1234'];

function getTargetSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'Timestamp', 'ID Bukti', 'Nama Siswa', 'Kelas', 'Nilai Objektif',
      'Skor PG', 'Skor BS', 'Status Uraian', 'Pelanggaran', 'Waktu Selesai', 'Payload_LKS_JSON'
    ]);
  }
  return sheet;
}

function doPost(e) {
  try {
    const sheet = getTargetSheet();
    const rawText = e.postData ? e.postData.contents : '{}';
    const data = JSON.parse(rawText);
    const lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      const existingIds = sheet.getRange(2, 2, lastRow - 1, 1).getValues().flat();
      if (existingIds.includes(data.idBukti)) {
        return ContentService.createTextOutput(JSON.stringify({ status: 'duplicate', idBukti: data.idBukti })).setMimeType(ContentService.MimeType.JSON);
      }
    }
    sheet.appendRow([
      new Date(), data.idBukti || '', data.nama || '', data.kelas || '',
      data.nilaiObjektif ?? data.skorAkhir ?? 0, data.detailPG || '', data.detailBS || '',
      data.detailUraian || '', data.pelanggaran ?? 0, data.waktuSelesai || '', JSON.stringify(data)
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', idBukti: data.idBukti })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const pin = (e && e.parameter && e.parameter.pin) ? String(e.parameter.pin).trim() : '';
  if (!TEACHER_PINS.includes(pin)) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'unauthorized', records: [] })).setMimeType(ContentService.MimeType.JSON);
  }
  const sheet = getTargetSheet();
  const lastRow = sheet.getLastRow();
  const records = [];
  if (lastRow > 1) {
    const jsonCol = sheet.getRange(2, 11, lastRow - 1, 1).getValues();
    for (let i = jsonCol.length - 1; i >= 0; i--) {
      if (!jsonCol[i][0]) continue;
      try { records.push(JSON.parse(jsonCol[i][0])); } catch (parseErr) {}
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', records: records })).setMimeType(ContentService.MimeType.JSON);
}
